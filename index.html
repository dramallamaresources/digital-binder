    <!-- Firebase Imports -->

    <script type="module">

        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";

        import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";

        import { getFirestore, doc, onSnapshot, setDoc, arrayUnion, arrayRemove, runTransaction, collection, setLogLevel } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";


        // Enable Firestore Debug Logging

        setLogLevel('Debug');


        // --- GLOBAL CONFIG & INITIALIZATION ---

        // NOTE: The variables __app_id, __firebase_config, and __initial_auth_token are provided by the canvas environment.

        // If running this file outside the canvas, you will need to replace them with real Firebase credentials

        // and an authentication method if you want to connect to a specific, permanent database.

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';

        const firebaseConfig = JSON.parse(typeof __firebase_config !== 'undefined' ? __firebase_config : '{}');

        const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;


        let app, db, auth, userId = null;

        let isAdmin = false; // Flag to determine if the user is the admin (i.e., authenticated via custom token)

        let binderData = { binderTitle: "My Digital Binder", sections: [] };

        let currentSectionId = null;


        const ui = {

            loadingSpinner: document.getElementById('loading-spinner'),

            mainUI: document.getElementById('main-ui'),

            modal: document.getElementById('custom-modal'),

            modalTitle: document.getElementById('modal-title'),

            modalMessage: document.getElementById('modal-message'),

            modalConfirmBtn: document.getElementById('modal-confirm-btn'),

            modalCancelBtn: document.getElementById('modal-cancel-btn'),

        };


        const BINDER_DOC_ID = 'main';


        // --- UTILITY FUNCTIONS ---


        /**

         * Determines the Firestore path for the public-facing binder data.

         * The data is public so viewers can read it, but it is keyed by the admin's UID.

         * @param {string} uid The admin's unique ID.

         * @returns {object} The Document Reference object.

         */

        const getBinderDocRef = (uid) => {

            const bindersCollection = collection(db, 'artifacts', appId, 'public', 'data', 'binders');

            return doc(bindersCollection, uid);

        };


        // --- AUTH & INITIALIZATION ---


        const initializeFirebase = async () => {

            try {

                app = initializeApp(firebaseConfig);

                db = getFirestore(app);

                auth = getAuth(app);


                // Start with loading

                ui.loadingSpinner.classList.remove('hidden');


                // 1. Try to sign in the Admin using the custom token

                if (initialAuthToken) {

                    await signInWithCustomToken(auth, initialAuthToken);

                    console.log('Admin signed in with custom token.');

                } else {

                    // 2. If no custom token, sign in anonymously (Public Viewer or un-privileged Admin)

                    await signInAnonymously(auth);

                    console.log('Signed in anonymously.');

                }


                // 3. Set up the Auth State Listener

                onAuthStateChanged(auth, (user) => {

                    if (user) {

                        userId = user.uid;

                        // Check if the current user ID matches the token-authenticated user ID

                        // This assumes only the user with the initialAuthToken should have edit rights

                        if (user.uid === (auth.currentUser?.uid || initialAuthToken)) {

                            isAdmin = !!initialAuthToken; // If we have a token, they are the admin

                        } else {

                            isAdmin = false;

                        }

                        console.log(`User ID: ${userId}, Is Admin: ${isAdmin}`);


                        // Determine the view and start listening to data

                        const params = new URLSearchParams(window.location.search);

                        const viewMode = params.get('view');

                        const targetAdminUid = params.get('uid');


                        if (viewMode === 'public' && targetAdminUid) {

                            // Public View Mode: use the UID from the URL parameter

                            startDataListener(targetAdminUid, 'public');

                        } else {

                            // Admin View Mode (or fallback)

                            startDataListener(userId, 'admin');

                        }


                    } else {

                        // User logged out/error

                        userId = null;

                        ui.loadingSpinner.classList.add('hidden');

                        ui.mainUI.innerHTML = '<p class="text-center text-red-500">Could not initialize application. Please ensure Firebase configuration is valid.</p>';

                        ui.mainUI.classList.remove('hidden');

                    }

                });


            } catch (error) {

                console.error("Firebase Initialization Error:", error);

                ui.loadingSpinner.classList.add('hidden');

                ui.mainUI.innerHTML = `<p class="text-center text-red-500">Initialization failed: ${error.message}</p>`;

                ui.mainUI.classList.remove('hidden');

            }

        };


        // --- FIRESTORE DATA LISTENER ---


        /**

         * Starts the real-time listener for the binder data.

         * @param {string} targetUid The UID of the admin whose binder we are reading.

         * @param {string} mode 'admin' or 'public'

         */

        const startDataListener = (targetUid, mode) => {

            const docRef = getBinderDocRef(targetUid);

            const unsubscribe = onSnapshot(docRef, (docSnap) => {

                if (docSnap.exists()) {

                    binderData = docSnap.data();

                    if (!binderData.sections || !Array.isArray(binderData.sections)) {

                         binderData.sections = []; // Ensure sections is an array

                    }

                    console.log("Binder Data Updated:", binderData);

                } else {

                    console.log("No existing binder data. Initializing a new one.");

                    // Initialize a default binder structure if it's the admin

                    if (mode === 'admin') {

                        binderData = {

                            binderTitle: "My Digital Binder",

                            sections: [{

                                id: crypto.randomUUID(),

                                title: "Welcome Section",

                                content: marked.parse("# Welcome to Your Digital Binder\n\nUse the panel to the left to edit, add, and reorder sections. This content is written in **Markdown**!")

                            }]

                        };

                        saveBinderData(); // Save the initial state

                    } else {

                        // Public view but no data found

                        binderData = {

                            binderTitle: "Binder Not Found",

                            sections: [{

                                id: 'error',

                                title: 'Error',

                                content: 'This binder link is invalid or the content has not been set yet.'

                            }]

                        };

                    }

                }


                // If no section is selected, default to the first one

                if (!currentSectionId && binderData.sections.length > 0) {

                    currentSectionId = binderData.sections[0].id;

                }


                renderUI(mode, targetUid);

            }, (error) => {

                console.error("Firestore Listener Error:", error);

                renderError(`Failed to load binder data: ${error.message}`);

            });


            // Clean up the listener on page unload (optional but good practice)

            window.addEventListener('beforeunload', unsubscribe);

        };


        // --- DATA MANIPULATION ---


        const saveBinderData = async (newBinderData = binderData) => {

             if (!isAdmin) {

                console.warn("Attempted to save data without admin privileges.");

                return;

            }


            const docRef = getBinderDocRef(userId);

            try {

                await setDoc(docRef, newBinderData, { merge: true });

                console.log("Binder data saved successfully.");

            } catch (error) {

                console.error("Error saving binder data:", error);

                renderError(`Failed to save data: ${error.message}`);

            }

        };


        const updateSectionContent = (sectionId, content) => {

            if (!isAdmin) return;

            const sectionIndex = binderData.sections.findIndex(s => s.id === sectionId);

            if (sectionIndex !== -1) {

                binderData.sections[sectionIndex].content = content;

                saveBinderData();

            }

        };


        const addSection = (title = "New Section") => {

            if (!isAdmin) return;

            const newSection = {

                id: crypto.randomUUID(),

                title: title,

                content: marked.parse("## Start Writing Here\n\nUse **Markdown** to format your document content.")

            };

            binderData.sections.push(newSection);

            currentSectionId = newSection.id;

            saveBinderData();

        };


        const deleteSection = async (sectionId) => {

            if (!isAdmin) return;


            const confirmed = await showModal('Delete Section', 'Are you sure you want to permanently delete this section? This cannot be undone.');


            if (confirmed) {

                binderData.sections = binderData.sections.filter(s => s.id !== sectionId);


                // If the deleted section was the active one, select the first available section

                if (currentSectionId === sectionId) {

                    currentSectionId = binderData.sections.length > 0 ? binderData.sections[0].id : null;

                }

                saveBinderData();

            }

        };


        const moveSection = (sectionId, direction) => {

            if (!isAdmin) return;

            const sections = binderData.sections;

            const index = sections.findIndex(s => s.id === sectionId);


            if (index === -1) return;


            let newIndex = index + (direction === 'up' ? -1 : 1);


            if (newIndex >= 0 && newIndex < sections.length) {

                // Swap elements

                [sections[index], sections[newIndex]] = [sections[newIndex], sections[index]];

                saveBinderData();

            }

        };


        // --- UI RENDERING ---


        const renderUI = (mode, targetUid) => {

            ui.loadingSpinner.classList.add('hidden');

            ui.mainUI.classList.remove('hidden');


            const selectedSection = binderData.sections.find(s => s.id === currentSectionId) || binderData.sections[0];


            if (mode === 'admin' && isAdmin) {

                renderAdminUI(selectedSection);

            } else {

                // Even if not strictly 'public' mode, if isAdmin is false, we render read-only

                renderPublicUI(selectedSection);

            }

        };


        const renderError = (message) => {

            ui.loadingSpinner.classList.add('hidden');

            ui.mainUI.classList.remove('hidden');

            ui.mainUI.innerHTML = `

                <div class="bg-red-100 border border-red-400 text-red-700 px-4 py-3 rounded relative" role="alert">

                    <strong class="font-bold">Error!</strong>

                    <span class="block sm:inline">${message}</span>

                </div>

            `;

        }


        const renderAdminUI = (selectedSection) => {

            const publicUrl = `${window.location.origin}${window.location.pathname}?view=public&uid=${userId}`;


            // --- Left Panel: Section Manager (Table of Contents) ---

            const sectionsHTML = binderData.sections.map((section, index) => {

                const isActive = section.id === currentSectionId;

                const canMoveUp = index > 0;

                const canMoveDown = index < binderData.sections.length - 1;


                return `

                    <div id="section-${section.id}" class="section-item flex items-center justify-between p-3 rounded-lg ${isActive ? 'active-section' : ''} transition duration-150 ease-in-out">

                        <div class="flex-grow min-w-0" onclick="handleSectionClick('${section.id}')">

                            <input type="text" value="${section.title}" 

                                onchange="handleTitleChange(event, '${section.id}')"

                                class="bg-transparent font-medium text-gray-800 w-full truncate border-none focus:ring-0 focus:border-blue-500 rounded p-1"

                                placeholder="Section Title">

                        </div>

                        <div class="flex space-x-1 ml-2 text-gray-500">

                            <button onclick="moveSection('${section.id}', 'up')" title="Move Up" ${canMoveUp ? '' : 'disabled'} class="p-1 rounded hover:bg-gray-200 disabled:opacity-30">

                                <svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M5 15l7-7 7 7" /></svg>

                            </button>

                            <button onclick="moveSection('${section.id}', 'down')" title="Move Down" ${canMoveDown ? '' : 'disabled'} class="p-1 rounded hover:bg-gray-200 disabled:opacity-30">

                                <svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 9l-7 7-7-7" /></svg>

                            </button>

                            <button onclick="deleteSection('${section.id}')" title="Delete Section" class="p-1 rounded hover:bg-red-200 text-red-500">

                                <svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16" /></svg>

                            </button>

                        </div>

                    </div>

                `;

            }).join('');


            // --- Right Panel: Editor ---

            const editorHTML = selectedSection ? `

                <h2 class="text-3xl font-bold text-gray-800 mb-4">${selectedSection.title} (Editing)</h2>

                <div class="mb-4">

                    <label for="section-content" class="block text-sm font-medium text-gray-700 mb-2">Content (Supports Markdown)</label>

                    <textarea id="section-content" class="w-full h-96 p-4 border border-gray-300 rounded-lg shadow-inner focus:ring-blue-500 focus:border-blue-500"

                        oninput="updatePreview()">${selectedSection.content}</textarea>

                </div>

                <div class="mb-6">

                    <h3 class="text-xl font-semibold text-gray-800 mb-2">Live Preview:</h3>

                    <div id="content-preview" class="p-4 bg-white border border-gray-200 rounded-lg min-h-[10rem] prose max-w-none">

                        ${marked.parse(selectedSection.content)}

                    </div>

                </div>

                <button onclick="saveContent()" class="w-full py-3 edit-button bg-blue-500 text-white font-semibold rounded-lg shadow-md hover:bg-blue-600 transition">

                    Save Content

                </button>

            ` : `<p class="text-center text-gray-500 py-10">Use the '+ Add Section' button to start building your binder.</p>`;



            ui.mainUI.innerHTML = `

                <div class="bg-white p-6 rounded-xl binder-shadow mb-6">

                    <div class="flex justify-between items-center mb-4">

                        <h2 class="text-xl font-semibold text-gray-800">Admin/Editor Mode - User ID: <span class="text-blue-500 font-mono text-sm break-all">${userId}</span></h2>

                        <button onclick="copyPublicLink('${publicUrl}')" class="px-4 py-2 bg-green-500 text-white font-medium rounded-lg hover:bg-green-600 transition flex items-center">

                            <svg xmlns="http://www.w3.org/2000/svg" class="h-5 w-5 mr-2" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M13.828 10.172a4 4 0 00-5.656 0l-4 4a4 4 0 105.656 5.656l1.102-1.101m-.758-4.881a4 4 0 005.656 0l4-4a4 4 0 00-5.656-5.656l-1.1 1.1"/></svg>

                            Copy Public Share Link

                        </button>

                    </div>

                    <p id="share-link-message" class="text-sm text-gray-500 mb-4 break-all">Shareable Viewer Link: <a href="${publicUrl}" target="_blank" class="text-blue-500 hover:underline">${publicUrl}</a></p>

                </div>


                <div class="flex flex-col lg:flex-row gap-8">

                    <!-- Left Panel: Navigation/TOC -->

                    <div class="lg:w-1/3 bg-white p-6 rounded-xl binder-shadow self-start sticky top-4">

                        <div class="flex justify-between items-center mb-4">

                            <h3 class="text-2xl font-semibold text-gray-800">Sections</h3>

                            <button onclick="addSection()" class="px-3 py-1 bg-blue-500 text-white font-medium rounded-lg hover:bg-blue-600 transition">

                                + Add Section

                            </button>

                        </div>

                        <div id="section-list" class="space-y-1">

                            ${sectionsHTML}

                        </div>

                    </div>


                    <!-- Right Panel: Content Editor -->

                    <div class="lg:w-2/3 bg-white p-6 rounded-xl binder-shadow">

                        ${editorHTML}

                    </div>

                </div>

            `;


            // Expose utility functions to the window for inline click handlers

            window.handleSectionClick = handleSectionClick;

            window.handleTitleChange = handleTitleChange;

            window.updatePreview = updatePreview;

            window.saveContent = saveContent;

            window.deleteSection = deleteSection;

            window.addSection = addSection;

            window.moveSection = moveSection;

            window.copyPublicLink = copyPublicLink;

        };


        const renderPublicUI = (selectedSection) => {

            const sectionsHTML = binderData.sections.map(section => {

                const isActive = section.id === currentSectionId;

                return `

                    <div id="section-${section.id}" class="section-item p-3 rounded-lg font-medium text-gray-800 ${isActive ? 'active-section' : ''}"

                         onclick="handleSectionClick('${section.id}')">

                        ${section.title}

                    </div>

                `;

            }).join('');


            const contentHTML = selectedSection ? marked.parse(selectedSection.content) : `

                <h2 class="text-3xl font-bold text-gray-800 mb-4">No Content</h2>

                <p class="text-gray-600">This binder has no sections yet.</p>

            `;


            ui.mainUI.innerHTML = `

                <div class="bg-blue-600 p-6 rounded-t-xl binder-shadow">

                    <h1 class="text-3xl font-bold text-white">${binderData.binderTitle}</h1>

                    <p class="text-blue-200 mt-1">Digital Binder Viewer Mode</p>

                </div>

                <div class="flex flex-col lg:flex-row gap-8 bg-white p-6 rounded-b-xl binder-shadow">

                    <!-- Left Panel: Navigation/TOC (Read-Only) -->

                    <div class="lg:w-1/4 bg-gray-50 p-4 rounded-lg self-start">

                        <h3 class="text-xl font-semibold text-gray-800 mb-4 border-b pb-2">Table of Contents</h3>

                        <div id="section-list" class="space-y-1">

                            ${sectionsHTML}

                        </div>

                    </div>


                    <!-- Right Panel: Content Viewer -->

                    <div class="lg:w-3/4">

                        <div class="prose max-w-none p-4">

                            ${contentHTML}

                        </div>

                    </div>

                </div>

            `;

            // Expose utility function for navigation

            window.handleSectionClick = handleSectionClick;

        };


        // --- HANDLERS ---


        const handleSectionClick = (sectionId) => {

            currentSectionId = sectionId;

            const params = new URLSearchParams(window.location.search);

            const mode = params.get('view') === 'public' ? 'public' : 'admin';

            const targetUid = params.get('uid') || userId;

            renderUI(mode, targetUid);

        };


        const handleTitleChange = (event, sectionId) => {

            if (!isAdmin) return;

            const newTitle = event.target.value.trim();

            const sectionIndex = binderData.sections.findIndex(s => s.id === sectionId);

            if (sectionIndex !== -1) {

                binderData.sections[sectionIndex].title = newTitle || "Untitled Section";

                saveBinderData();

            }

        };


        const updatePreview = () => {

            const content = document.getElementById('section-content').value;

            const previewElement = document.getElementById('content-preview');

            if (previewElement) {

                // Use a safe rendering library like marked.js

                previewElement.innerHTML = marked.parse(content);

            }

        };


        const saveContent = () => {

            if (!isAdmin) return;

            const content = document.getElementById('section-content').value;

            if (currentSectionId) {

                // This updates the local data and triggers a save to Firestore

                updateSectionContent(currentSectionId, content);

            }

            // A message is not explicitly needed as onSnapshot updates the UI automatically

        };


        const copyPublicLink = (link) => {

            try {

                // Use the older execCommand for better iFrame compatibility

                const textarea = document.createElement('textarea');

                textarea.value = link;

                document.body.appendChild(textarea);

                textarea.select();

                document.execCommand('copy');

                document.body.removeChild(textarea);


                const message = document.getElementById('share-link-message');

                message.innerHTML = `Link copied! <span class="text-green-500 font-semibold"> (Copied!)</span>`;

                setTimeout(() => {

                    message.innerHTML = `Shareable Viewer Link: <a href="${link}" target="_blank" class="text-blue-500 hover:underline">${link}</a>`;

                }, 2000);


            } catch (err) {

                console.error('Failed to copy text: ', err);

            }

        };


        // --- CUSTOM MODAL IMPLEMENTATION (for alert/confirm) ---


        let resolveModalPromise;


        const showModal = (title, message) => {

            return new Promise((resolve) => {

                resolveModalPromise = resolve;

                ui.modalTitle.textContent = title;

                ui.modalMessage.textContent = message;

                ui.modal.classList.remove('hidden');

                ui.modal.classList.add('flex');

            });

        };


        ui.modalConfirmBtn.addEventListener('click', () => {

            ui.modal.classList.add('hidden');

            ui.modal.classList.remove('flex');

            resolveModalPromise(true);

        });


        ui.modalCancelBtn.addEventListener('click', () => {

            ui.modal.classList.add('hidden');

            ui.modal.classList.remove('flex');

            resolveModalPromise(false);

        });


        // --- START APPLICATION ---

        initializeFirebase();

    </script>
